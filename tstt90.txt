        function toPersianNumber(number) {
            if (number === undefined || number === null) return '';
            
            const numStr = String(number);
            let result = '';
            
            for (let i = 0; i < numStr.length; i++) {
                const char = numStr[i];
                if (char >= '0' && char <= '9') {
                    result += PERSIAN_NUMBERS[parseInt(char)];
                } else {
                    result += char;
                }
            }
            
            return result;
        }

        function toEnglishNumber(persianNumber) {
            if (!persianNumber) return '';
            
            const numStr = String(persianNumber);
            let result = '';
            
            for (let i = 0; i < numStr.length; i++) {
                const char = numStr[i];
                const persianIndex = PERSIAN_NUMBERS.indexOf(char);
                const arabicIndex = ARABIC_NUMBERS.indexOf(char);
                
                if (persianIndex !== -1) {
                    result += persianIndex;
                } else if (arabicIndex !== -1) {
                    result += arabicIndex;
                } else {
                    result += char;
                }
            }
            
            return result;
        }

        function numberWithCommas(x) {
            if (x === undefined || x === null) return '0';
            const num = parseFloat(x);
            if (isNaN(num)) return '0';
            const parts = num.toString().split('.');
            parts[0] = parts[0].replace(/\B(?=(\d{3})+(?!\d))/g, ",");
            return parts.join(".");
        }

        function convertAmount(amount, fromCurrency, toCurrency) {
            if (fromCurrency === toCurrency) return amount;
            const amountInToman = amount * currencyRates[fromCurrency].rate;
            return amountInToman / currencyRates[toCurrency].rate;
        }

        function formatCurrency(amount) {
            const rateInfo = currencyRates[currency];
            const convertedAmount = convertAmount(amount, 'toman', currency);
            const formattedAmount = toPersianNumber(numberWithCommas(convertedAmount.toFixed(rateInfo.decimals)));
            return formattedAmount + ' ' + rateInfo.symbol;
        }

        function gregorianToJalali(gy, gm, gd) {
            const g_days_in_month = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
            const j_days_in_month = [31, 31, 31, 31, 31, 31, 30, 30, 30, 30, 30, 29];
            
            let gy2 = (gm > 2) ? (gy + 1) : gy;
            let days = 355666 + (365 * gy) + Math.floor((gy2 + 3) / 4) - Math.floor((gy2 + 99) / 100) + Math.floor((gy2 + 399) / 400) + gd;
            
            for (let i = 0; i < gm - 1; i++) {
                days += g_days_in_month[i];
            }
            
            if (gm > 2 && ((gy % 4 === 0 && gy % 100 !== 0) || (gy % 400 === 0))) {
                days++;
            }
            
            let jy = -1595 + (33 * Math.floor(days / 12053));
            days %= 12053;
            jy += 4 * Math.floor(days / 1461);
            days %= 1461;
            
            if (days > 365) {
                jy += Math.floor((days - 1) / 365);
                days = (days - 1) % 365;
            }
            
            let jm, jd;
            if (days < 186) {
                jm = 1 + Math.floor(days / 31);
                jd = 1 + (days % 31);
            } else {
                days -= 186;
                jm = 7 + Math.floor(days / 30);
                jd = 1 + (days % 30);
            }
            
            return [jy, jm, jd];
        }

        function jalaliToGregorian(jy, jm, jd) {
            const g_days_in_month = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
            const j_days_in_month = [31, 31, 31, 31, 31, 31, 30, 30, 30, 30, 30, 29];
            
            jy += 1595;
            let days = -355668 + (365 * jy) + (Math.floor(jy / 33) * 8) + Math.floor(((jy % 33) + 3) / 4);
            
            for (let i = 0; i < jm - 1; i++) {
                days += j_days_in_month[i];
            }
            
            days += jd;
            
            let gy = 400 * Math.floor(days / 146097);
            days %= 146097;
            
            if (days > 36524) {
                gy += 100 * Math.floor(--days / 36524);
                days %= 36524;
                if (days >= 365) days++;
            }
            
            gy += 4 * Math.floor(days / 1461);
            days %= 1461;
            
            if (days > 365) {
                gy += Math.floor((days - 1) / 365);
                days = (days - 1) % 365;
            }
            
            let gd = days + 1;
            let gm;
            
            for (gm = 0; gm < 12; gm++) {
                let v = g_days_in_month[gm];
                if (gm === 1 && ((gy % 4 === 0 && gy % 100 !== 0) || (gy % 400 === 0))) {
                    v++;
                }
                if (gd <= v) break;
                gd -= v;
            }
            
            return [gy, gm + 1, gd];
        }

        function checkPassword() {
            safeLoadData();
            
            const password = localStorage.getItem('journalPassword');
            
            if (!password || password === 'null' || password === 'undefined' || password === '' || password === '0000') {
                if (password === '0000') {
                    localStorage.removeItem('journalPassword');
                }
                showMainContent();
                return;
            }
            
            document.getElementById('lockScreen').style.display = 'flex';
            
            setTimeout(function() {
                var passwordInput = document.getElementById('passwordInput');
                if (passwordInput) {
                    passwordInput.focus();
                }
            }, 100);
        }

        function handlePasswordEnter(event) {
            event = event || window.event;
            if (event.key === 'Enter' || event.keyCode === 13) {
                unlockJournal();
            }
        }

        function unlockJournal() {
            const passwordInput = document.getElementById('passwordInput');
            if (!passwordInput) return;
            
            const enteredPassword = passwordInput.value;
            const savedPassword = localStorage.getItem('journalPassword');
            
            if (enteredPassword.length !== 4) {
                showPasswordError('❌ رمز عبور باید ۴ رقمی باشد');
                return;
            }
            
            if (!savedPassword || savedPassword === 'null' || savedPassword === 'undefined' || savedPassword === '') {
                if (enteredPassword === '0000') {
                    localStorage.removeItem('journalPassword');
                    showMainContent();
                } else {
                    localStorage.setItem('journalPassword', enteredPassword);
                    showMainContent();
                }
                return;
            }
            
            if (enteredPassword === savedPassword) {
                showMainContent();
            } else {
                showPasswordError('❌ رمز عبور نادرست است');
                passwordInput.value = '';
                
                setTimeout(function() {
                    if (passwordInput) {
                        passwordInput.focus();
                    }
                }, 100);
            }
        }

        function showPasswordError(message) {
            const errorElement = document.getElementById('errorMessage');

                errorElement.style.color = '#e74c3c';
                errorElement.style.fontWeight = 'bold';
                
                setTimeout(function() {
                    errorElement.textContent = '';
                }, 3000);
            }
        }

        function togglePasswordVisibility() {

            const toggleBtn = document.getElementById('togglePasswordBtn');

                toggleBtn.innerHTML = '👁️';

            }
        }

        function safeSaveData() {

                localStorage.setItem('journalCategories', JSON.stringify(categories));

        function safeLoadData() {
            try {
                var savedEntries = localStorage.getItem('simpleJournal');
                if (savedEntries) {
                    var parsed = JSON.parse(savedEntries);
                    if (Array.isArray(parsed)) {
                        entries = parsed;
                    }
                }
                
                var savedRow = localStorage.getItem('currentRow');
                if (savedRow) {
                    currentRow = parseInt(savedRow, 10) || 1;
                }
                
                var savedCategories = localStorage.getItem('journalCategories');
                if (savedCategories) {
                    try {
                        var parsedCategories = JSON.parse(savedCategories);
                        if (Array.isArray(parsedCategories)) {
                            categories = parsedCategories;
                        }
                    } catch (e) {
                        console.warn('خطا در بارگذاری دسته‌بندی‌ها:', e);
                    }
                } else {
                    categories = defaultCategoriesByLang[currentLanguage] || defaultCategoriesByLang['fa'];
                }
                
                var savedWords = localStorage.getItem('suggestedWords');
                if (savedWords) {
                    try {
                        var parsedWords = JSON.parse(savedWords);
                        if (Array.isArray(parsedWords)) {
                            suggestedWords = parsedWords;
                        }
                    } catch (e) {
                        console.warn('خطا در بارگذاری کلمات پیشنهادی:', e);
                    }
                } else {
                    suggestedWords = defaultWordsByLang[currentLanguage] || defaultWordsByLang['fa'];
                }
                
                var savedCurrency = localStorage.getItem('journalCurrency');
                if (savedCurrency) {
                    currency = savedCurrency;
                }
                
                var savedCurrencyRates = localStorage.getItem('currencyRates');
                if (savedCurrencyRates) {
                    try {
                        var parsedRates = JSON.parse(savedCurrencyRates);
                        currencyRates = parsedRates;
                    } catch (e) {
                        console.warn('خطا در بارگذاری نرخ ارز:', e);
                    }
                }
                
                var savedHistory = localStorage.getItem('entryHistory');
                if (savedHistory) {
                    try {
                        entryHistory = JSON.parse(savedHistory);
                    } catch (e) {
                        console.warn('خطا در بارگذاری تاریخچه ویرایش‌ها:', e);
                        entryHistory = {};
                    }
                } else {
                    entryHistory = {};
                }
                
            } catch (error) {
                console.error('خطا در بارگذاری داده‌ها:', error);
                entries = [];
                currentRow = 1;
                categories = defaultCategoriesByLang[currentLanguage] || defaultCategoriesByLang['fa'];
                suggestedWords = defaultWordsByLang[currentLanguage] || defaultWordsByLang['fa'];
                entryHistory = {};
            }
        }

        function updateDisplay() {

                var displayEntries = searchResults.length > 0 ? searchResults : entries;

                var endIndex = startIndex + recordsPerPage;
                var pageEntries = displayEntries.slice(startIndex, endIndex);

                var currentPageEl = document.getElementById('currentPage');

                if (prevBtn) prevBtn.disabled = currentPage <= 1;
                if (nextBtn) nextBtn.disabled = currentPage >= totalPages;

                if (pageEntries.length === 0) {
                    journalBody.innerHTML = '<tr><td colspan="7" style="text-align: center; padding: 20px; color: #7f8c8d;">' + t('noRecords') + '</td></tr>';
                    return;
                }

                var fragment = document.createDocumentFragment();
                
                for (var i = 0; i < pageEntries.length; i++) {
                    var entry = pageEntries[i];
                    var row = document.createElement('tr');
                    row.setAttribute('data-row', entry.row);
                    
                    var amountDisplay = formatCurrency(entry.amount);
                    var paymentCell = '';
                    var receiptCell = '';
                    
                    if (entry.type === 'expense' || entry.type === 'credit') {
                        paymentCell = '<td class="expense">' + amountDisplay + '</td>';
                        receiptCell = '<td></td>';
                    } else {
                        paymentCell = '<td></td>';
                        receiptCell = '<td class="income">' + amountDisplay + '</td>';
                    }
                    
                    // بررسی آیا این رکورد ویرایش شده است
                    const hasHistory = entryHistory[entry.row] && entryHistory[entry.row].length > 0;
                    const starIcon = hasHistory ? '<span class="correction-star" onclick="showOriginalRecord(' + entry.row + ', this)" aria-label="نمایش نسخه قبلی">⭐</span>' : '';
                    
                    row.innerHTML = 
                        '<td>' + toPersianNumber(entry.row) + starIcon + '</td>' +
                        '<td>' + formatDate(entry.date) + '</td>' +
                        '<td style="text-align: right; padding-right: 15px;">' + entry.description + '</td>' +
                        paymentCell +
                        receiptCell +
                        '<td>' + entry.category + '</td>' +
                        '<td>' +
                        '<button class="edit-btn" onclick="editEntry(' + entry.row + ')" title="' + t('edit') + '" aria-label="ویرایش رکورد">✏️</button>' +
                        '<button class="delete-btn" onclick="deleteSingleEntry(' + entry.row + ')" title="حذف" aria-label="حذف رکورد" style="margin-right: 5px; background: #e74c3c; color: white; border: none; padding: 2px 6px; border-radius: 3px; cursor: pointer;">🗑️</button>' +
                        '</td>';
                    
                    // اگر ویرایش شده باشد، کلاس اضافه کن

                        row.classList.add('corrected-record');
                    }
                    
                    fragment.appendChild(row);
                }
                
                journalBody.appendChild(fragment);
                
            } catch (error) {
                console.error('خطا در updateDisplay:', error);
            }
        }

        function formatDate(dateString) {
            try {
                if (!dateString) return 'تاریخ نامعتبر';
                
                var parts = dateString.split('/');
                if (parts.length !== 3) {
                    return dateString;
                }
                
                var year = parseInt(parts[0]);
                var month = parseInt(parts[1]);
                var day = parseInt(parts[2]);
                
                if (isNaN(year) || isNaN(month) || isNaN(day)) {
                    return 'تاریخ نامعتبر';
                }
                
                if (dateFormat === 'gregorian') {
                    if (year < 1500) {
                        var gregorian = jalaliToGregorian(year, month, day);
                        year = gregorian[0];
                        month = gregorian[1];
                        day = gregorian[2];
                    }
                    
                    return toPersianNumber(year) + '/' + 
                           toPersianNumber(month.toString().padStart(2, '0')) + '/' + 
                           toPersianNumber(day.toString().padStart(2, '0'));
                } else {
                    if (year > 1500) {
                        var jalali = gregorianToJalali(year, month, day);
                        year = jalali[0];
                        month = jalali[1];
                        day = jalali[2];
                    }
                    
                    var monthNames = ['فروردین', 'اردیبهشت', 'خرداد', 'تیر', 'مرداد', 'شهریور',
                                     'مهر', 'آبان', 'آذر', 'دی', 'بهمن', 'اسفند'];
                    
                    var monthName = monthNames[month - 1] || 'ماه';
                    
                    return toPersianNumber(year) + ' ' + monthName + ' ' + toPersianNumber(day);
                }
            } catch (error) {
                console.error('خطا در فرمت تاریخ:', error);
                return dateString || 'تاریخ نامعتبر';
            }
        }

        // ==================== سیستم ویرایش با علامت ستاره ====================
        function editEntry(row) {
            // ذخیره نسخه قبلی قبل از ویرایش
            for (var i = 0; i < entries.length; i++) {
                if (entries[i].row === row) {
                    if (!entryHistory[row]) {
                        entryHistory[row] = [];
                    }
                    
                    // ذخیره نسخه قبلی در تاریخچه
                    entryHistory[row].push({
                        ...entries[i],
                        changedAt: new Date().toISOString()
                    });
                    
                    break;
                }
            }
            
            showEditModal(row);
        }

        function showOriginalRecord(row, starElement) {
            const history = entryHistory[row];
            if (!history || history.length === 0) return;
            
            // آخرین نسخه قبل از ویرایش
            const lastOriginal = history[history.length - 1];
            
            // ایجاد یا به‌روزرسانی نمایش نسخه اصلی
            let originalDisplay = starElement.nextElementSibling;
            if (!originalDisplay || !originalDisplay.classList.contains('original-record')) {
                originalDisplay = document.createElement('div');
                originalDisplay.className = 'original-record';
                starElement.parentNode.insertBefore(originalDisplay, starElement.nextSibling);
            }
            
            originalDisplay.innerHTML = `
                <strong>نسخه قبلی:</strong><br>
                شرح: ${lastOriginal.description}<br>
                مبلغ: ${formatCurrency(lastOriginal.amount)}<br>
                نوع: ${lastOriginal.type === 'income' ? 'دریافت' : 
                        lastOriginal.type === 'expense' ? 'پرداخت' : 
                        lastOriginal.type === 'debt' ? 'طلب' : 'بدهی'}<br>
                تاریخ ویرایش: ${new Date(lastOriginal.changedAt).toLocaleDateString('fa-IR')}
            `;
            
            originalDisplay.classList.toggle('show');
        }

        // ==================== توابع خلاصه حساب‌ها ====================
        function updateBalance() {
            var totalIncome = 0;
            var totalExpense = 0;
            var totalDebt = 0;
            var totalCredit = 0;
            
            for (var i = 0; i < entries.length; i++) {
                var entry = entries[i];
                if (entry.type === 'income') totalIncome += entry.amount;
                else if (entry.type === 'expense') totalExpense += entry.amount;
                else if (entry.type === 'debt') totalDebt += entry.amount;
                else if (entry.type === 'credit') totalCredit += entry.amount;
            }
            
            var totalReceipts = totalIncome + totalDebt;
            var totalPayments = totalExpense + totalCredit;
            var balance = totalReceipts - totalPayments;
            
            // به‌روزرسانی خلاصه افقی
            var horizontalExpense = document.getElementById('horizontalExpense');
            var horizontalIncome = document.getElementById('horizontalIncome');
            var horizontalBalance = document.getElementById('horizontalBalance');
            
            if (horizontalExpense) {
                horizontalExpense.textContent = formatCurrency(totalPayments);
            }
            if (horizontalIncome) {
                horizontalIncome.textContent = formatCurrency(totalReceipts);
            }
            if (horizontalBalance) {
                horizontalBalance.textContent = formatCurrency(Math.abs(balance));
                horizontalBalance.className = 'amount ' + (balance >= 0 ? 'income' : 'expense');
            }
        }

                var now = new Date();
                updateTodayDate();

                if (timeElement) {

                    }

            }
            updateClock();
            setInterval(updateClock, 1000);
        }

        // ==================== سیستم پیشنهاد کلمات ====================
        function toggleSuggestions() {
            var suggestions = document.getElementById('descriptionSuggestions');
            var arrow = document.querySelector('.dropdown-arrow');
            
            if (!suggestions || !arrow) return;
            
            if (suggestions.classList.contains('show')) {
                suggestions.classList.remove('show');
                arrow.classList.remove('rotated');
                arrow.setAttribute('aria-label', 'نمایش پیشنهادات');
            } else {
                showSuggestions();
                arrow.classList.add('rotated');
                arrow.setAttribute('aria-label', 'مخفی کردن پیشنهادات');
            }
        }

        function showSuggestions() {
            var input = document.getElementById('editDescription');
            var suggestions = document.getElementById('descriptionSuggestions');
            
            if (!input || !suggestions) return;
            
            var searchTerm = input.value.toLowerCase();
            currentSuggestions = [];
            
            if (searchTerm === '') {
                currentSuggestions = suggestedWords.slice(0, 10);
            } else {
                for (var i = 0; i < suggestedWords.length; i++) {
                    if (suggestedWords[i].toLowerCase().includes(searchTerm)) {
                        currentSuggestions.push(suggestedWords[i]);
                        if (currentSuggestions.length >= 10) break;
                    }
                }
            }
            
            suggestions.innerHTML = '';
            
            if (currentSuggestions.length === 0) {
                suggestions.innerHTML = '<div class="suggestion-item">' + t('noResults') + '</div>';
            } else {
                for (var i = 0; i < currentSuggestions.length; i++) {
                    var item = document.createElement('div');
                    item.className = 'suggestion-item';
                    item.textContent = currentSuggestions[i];
                    item.setAttribute('data-index', i);
                    item.setAttribute('role', 'option');
                    item.onclick = function() {
                        selectSuggestion(parseInt(this.getAttribute('data-index')));
                    };
                    item.onmouseover = function() {
                        setActiveSuggestion(parseInt(this.getAttribute('data-index')));
                    };
                    suggestions.appendChild(item);
                }
            }
            
            suggestions.classList.add('show');
            selectedSuggestionIndex = -1;
        }

        function hideSuggestions() {
            var suggestions = document.getElementById('descriptionSuggestions');
            var arrow = document.querySelector('.dropdown-arrow');
            
            if (suggestions) {
                suggestions.classList.remove('show');
            }
            if (arrow) {
                arrow.classList.remove('rotated');
                arrow.setAttribute('aria-label', 'نمایش پیشنهادات');
            }
            selectedSuggestionIndex = -1;
        }

        function setActiveSuggestion(index) {
            var items = document.querySelectorAll('.suggestion-item');
            items.forEach(function(item, i) {
                if (i === index) {
                    item.classList.add('active');
                    item.setAttribute('aria-selected', 'true');
                } else {
                    item.classList.remove('active');
                    item.setAttribute('aria-selected', 'false');
                }
            });
            selectedSuggestionIndex = index;
        }

        function selectSuggestion(index) {
            if (index >= 0 && index < currentSuggestions.length) {
                document.getElementById('editDescription').value = currentSuggestions[index];
                hideSuggestions();
            }
        }

        function changeLanguage(lang) {

            localStorage.setItem('journalLanguage', lang);

            const htmlElement = document.querySelector('html');
            if (htmlElement) {
                if (lang === 'ar' || lang === 'fa' || lang === 'ku') {
                    htmlElement.setAttribute('dir', 'rtl');
                } else {
                    htmlElement.setAttribute('dir', 'ltr');
                }
            }
            
            const savedWords = localStorage.getItem('suggestedWords');
            if (!savedWords || savedWords === '[]') {
                suggestedWords = defaultWordsByLang[lang] || defaultWordsByLang['fa'];
                localStorage.setItem('suggestedWords', JSON.stringify(suggestedWords));

            const savedCategories = localStorage.getItem('journalCategories');

            updateHorizontalSummary();
            updateCategoriesDropdown();
            updateDateFormatDisplay();
            updateCurrencyDisplay();

            if (langDisplay) {

                    'fa': 'فارسی',

                    'ku': 'کوردی',
                    'ar': 'عربی'
                };

                option.classList.remove('active');

        function updateCurrencyPreview() {
            var currencySelect = document.getElementById('currencySelect');
            var currencyRate = document.getElementById('currencyRate');
            var preview = document.getElementById('currencyPreview');
            
            if (!currencySelect || !currencyRate || !preview) return;
            
            var selectedCurrency = currencySelect.value;
            var rate = parseFloat(currencyRate.value) || 1;
            
            var exampleAmount = 100000;
            var convertedAmount = exampleAmount / rate;
            
            preview.textContent = numberWithCommas(exampleAmount) + ' تومان = ' +

        function saveCurrencySettings() {
            var currencySelect = document.getElementById('currencySelect');

        function switchReportTab(tabName) {
            // مخفی کردن همه تب‌ها
            document.querySelectorAll('.report-tab').forEach(tab => {
                tab.style.display = 'none';
                tab.setAttribute('aria-hidden', 'true');
            });
            
            // غیرفعال کردن همه دکمه‌ها
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('active');
                btn.setAttribute('aria-selected', 'false');
            });
            
            // نمایش تب انتخاب شده
            const selectedTab = document.getElementById(tabName + 'Report');
            const selectedBtn = document.querySelector(`.tab-btn[onclick*="${tabName}"]`);
            
            if (selectedTab) {
                selectedTab.style.display = 'block';
                selectedTab.setAttribute('aria-hidden', 'false');
            }
            if (selectedBtn) {
                selectedBtn.classList.add('active');
                selectedBtn.setAttribute('aria-selected', 'true');
            }
            
            // تولید گزارش مربوطه
            switch(tabName) {
                case 'category':
                    generateCategoryReport();
                    break;
                case 'type':
                    generateTypeReport();
                    break;
            }
        }

        function generateCategoryReport() {
            const stats = {};
            
            // محاسبه آمار دسته‌بندی‌ها
            entries.forEach(entry => {
                if (!stats[entry.category]) {
                    stats[entry.category] = {
                        income: 0,
                        expense: 0,
                        count: 0
                    };
                }
                
                if (entry.type === 'income' || entry.type === 'debt') {
                    stats[entry.category].income += entry.amount;
                } else {
                    stats[entry.category].expense += entry.amount;
                }
                stats[entry.category].count++;
            });
            
            // نمایش آمار
            const statsElement = document.getElementById('categoryStats');
            if (statsElement) {
                let html = '<h4>آمار دسته‌بندی‌ها:</h4><ul style="list-style: none; padding: 0;">';
                
                Object.keys(stats).forEach(category => {
                    const total = stats[category].income + stats[category].expense;
                    html += `<li style="padding: 5px 0; border-bottom: 1px solid #eee;">
                        <strong>${category}:</strong> ${stats[category].count} تراکنش - 
                        کل مبلغ: ${formatCurrency(total)}
                    </li>`;
                });
                
                html += '</ul>';
                statsElement.innerHTML = html;
            }
        }

                credit: { amount: 0, count: 0 }

        function exportToExcel() {
            // محاسبه خلاصه
            let totalIncome = 0, totalExpense = 0, totalDebt = 0, totalCredit = 0;
            entries.forEach(entry => {
                if (entry.type === 'income') totalIncome += entry.amount;
                else if (entry.type === 'expense') totalExpense += entry.amount;
                else if (entry.type === 'debt') totalDebt += entry.amount;
                else if (entry.type === 'credit') totalCredit += entry.amount;
            });
            
            const totalReceipts = totalIncome + totalDebt;
            const totalPayments = totalExpense + totalCredit;
            const balance = totalReceipts - totalPayments;
            
            // ایجاد داده CSV
            let csvData = "ردیف,تاریخ,شرح,نوع,مبلغ,دسته‌بندی,واحد پول\n";
            
            entries.forEach(entry => {
                const typeText = 
                    entry.type === 'income' ? 'دریافت' :
                    entry.type === 'expense' ? 'پرداخت' :
                    entry.type === 'debt' ? 'طلب' : 'بدهی';
                
                csvData += `${entry.row},"${formatDate(entry.date)}","${entry.description}",${typeText},${entry.amount},${entry.category},${currencyRates[currency].symbol}\n`;
            });
            
            // اضافه کردن خلاصه
            csvData += `\nخلاصه حساب‌ها\n`;
            csvData += `جمع پرداخت‌ها,${totalPayments}\n`;
            csvData += `جمع دریافت‌ها,${totalReceipts}\n`;
            csvData += `مانده حساب,${balance}\n`;
            csvData += `تعداد کل رکوردها,${entries.length}\n`;
            csvData += `تاریخ خروجی,${new Date().toLocaleDateString('fa-IR')}\n`;
            
            // ایجاد فایل CSV
            const blob = new Blob([csvData], { type: 'text/csv;charset=utf-8;' });
            const url = URL.createObjectURL(blob);
            const link = document.createElement('a');
            const today = new Date();
            const fileName = `دفتر-روزنامه-${today.getFullYear()}-${today.getMonth()+1}-${today.getDate()}.csv`;
            
            link.href = url;
            link.setAttribute('download', fileName);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            
            showNotification(t('exportExcelSuccess'), 'success');
        }

        function performSearch() {
            var searchQuery = document.getElementById('searchQuery');
            if (!searchQuery) return;
            
            var query = searchQuery.value.trim().toLowerCase();
            
            if (!query) {
                showNotification('لطفاً عبارت جستجو را وارد کنید.', 'error');
                return;
            }
            
            currentSearchQuery = query;
            searchResults = [];
            
            for (var i = 0; i < entries.length; i++) {
                var entry = entries[i];
                if (entry.description.toLowerCase().includes(query) ||
                    entry.category.toLowerCase().includes(query) ||
                    entry.type.toLowerCase().includes(query) ||
                    entry.amount.toString().includes(query)) {
                    searchResults.push(entry);
                }
            }
            
            currentPage = 1;
            updateDisplay();
            
            var resultsElement = document.getElementById('searchResults');
            if (resultsElement) {
                if (searchResults.length === 0) {
                    resultsElement.innerHTML = '<div class="search-summary"><h4>' + t('searchResults') + '</h4><p>' + t('noResults') + '</p></div>';
                } else {
                    resultsElement.innerHTML = '<div class="search-summary"><h4>' + t('searchResults') + '</h4><p>' + searchResults.length + ' ' + t('resultsFound') + '</p></div>';
                }
            }
        }

        function changePage(direction) {
            var totalPages = Math.ceil((searchResults.length > 0 ? searchResults : entries).length / recordsPerPage);
            currentPage += direction;
            if (currentPage < 1) currentPage = 1;
            if (currentPage > totalPages) currentPage = totalPages;
            updateDisplay();
        }

        function saveNewPassword() {
            var currentPasswordInput = document.getElementById('currentPasswordInput');
            var newPasswordInput = document.getElementById('newPasswordInput');
            var confirmPasswordInput = document.getElementById('confirmPasswordInput');
            
            if (!currentPasswordInput || !newPasswordInput || !confirmPasswordInput) return;
            
            var currentPassword = currentPasswordInput.value;
            var newPassword = newPasswordInput.value;
            var confirmPassword = confirmPasswordInput.value;
            
            if (newPassword.length !== 4) {
                showNotification('❌ رمز جدید باید ۴ رقمی باشد', 'error');
                return;
            }
            
            if (newPassword !== confirmPassword) {
                showNotification('❌ رمز جدید و تکرار آن مطابقت ندارند', 'error');
                return;
            }
            
            var savedPassword = localStorage.getItem('journalPassword');
            
            if (savedPassword && savedPassword !== 'null' && savedPassword !== 'undefined' && savedPassword !== '') {
                if (currentPassword !== savedPassword) {
                    showNotification('❌ رمز عبور فعلی نادرست است', 'error');
                    return;
                }
            }
            
            localStorage.setItem('journalPassword', newPassword);
            hidePasswordSettings();
            showNotification('✅ رمز عبور با موفقیت تغییر کرد', 'success');
        }

        function restoreCompleteBackup() {
            if (!selectedBackupFile) {
                showNotification('لطفاً ابتدا یک فایل پشتیبان انتخاب کنید', 'error');
                return;
            }
            
            if (!confirm('⚠️ هشدار!\nبازگردانی داده‌ها تمام اطلاعات فعلی را حذف می‌کند.\nآیا مطمئن هستید؟')) {
                return;
            }
            
            var reader = new FileReader();
            reader.onload = function(e) {
                try {
                    var data = JSON.parse(e.target.result);
                    
                    entries = data.entries || [];
                    currentRow = data.currentRow || 1;
                    categories = data.categories || (defaultCategoriesByLang[currentLanguage] || defaultCategoriesByLang['fa']);
                    suggestedWords = data.suggestedWords || (defaultWordsByLang[currentLanguage] || defaultWordsByLang['fa']);
                    journalName = data.journalName || t('journalTitle');
                    journalSubtitle = data.journalSubtitle || t('journalSubtitle');
                    entryHistory = data.entryHistory || {};
                    
                    if (data.currency) {
                        currency = data.currency;
                    }
                    
                    if (data.currencyRates) {
                        currencyRates = data.currencyRates;
                    }
                    
                    safeSaveData();
                    localStorage.setItem('journalCategories', JSON.stringify(categories));
                    localStorage.setItem('suggestedWords', JSON.stringify(suggestedWords));
                    localStorage.setItem('journalName', journalName);
                    localStorage.setItem('journalSubtitle', journalSubtitle);
                    localStorage.setItem('journalCurrency', currency);
                    localStorage.setItem('currencyRates', JSON.stringify(currencyRates));
                    localStorage.setItem('entryHistory', JSON.stringify(entryHistory));
                    
                    document.getElementById('journalTitle').textContent = journalName;
                    document.getElementById('journalSubtitle').textContent = journalSubtitle;
                    updateDisplay();
                    updateBalance();
                    updateCategoriesDropdown();
                    updateCurrencyDisplay();
                    
                    hideRestoreModal();
                    showNotification('✅ داده‌ها با موفقیت بازگردانی شدند', 'success');
                    
                } catch (error) {
                    console.error('خطا در بازگردانی:', error);
                    showNotification('❌ خطا در بازگردانی داده‌ها. فایل معتبر نیست', 'error');
                }
            };
            
            reader.readAsText(selectedBackupFile);
        }

        function createCompleteBackup() {
            var data = {
                entries: entries,
                currentRow: currentRow,
                categories: categories,
                suggestedWords: suggestedWords,
                journalName: journalName,
                journalSubtitle: journalSubtitle,
                currency: currency,
                currencyRates: currencyRates,
                entryHistory: entryHistory,
                version: DATA_VERSION,
                exportDate: new Date().toISOString()
            };
            
            var dataStr = JSON.stringify(data, null, 2);
            var dataUri = 'data:application/json;charset=utf-8,'+ encodeURIComponent(dataStr);
            
            var exportFileDefaultName = 'دفتر-روزنامه-' + new Date().toISOString().slice(0,10) + '.json';
            
            var linkElement = document.createElement('a');
            linkElement.setAttribute('href', dataUri);
            linkElement.setAttribute('download', exportFileDefaultName);
            linkElement.click();
            
            showNotification('✅ داده‌ها با موفقیت صادر شدند', 'success');
        }

        }

                descInput.addEventListener('keydown', function(e) {
                    var suggestions = document.getElementById('descriptionSuggestions');
                    if (!suggestions || !suggestions.classList.contains('show')) return;
                    
                    if (e.key === 'ArrowDown') {
                        e.preventDefault();
                        var newIndex = selectedSuggestionIndex + 1;
                        if (newIndex >= currentSuggestions.length) newIndex = 0;
                        setActiveSuggestion(newIndex);
                    } else if (e.key === 'ArrowUp') {
                        e.preventDefault();
                        var newIndex = selectedSuggestionIndex - 1;
                        if (newIndex < 0) newIndex = currentSuggestions.length - 1;
                        setActiveSuggestion(newIndex);
                    } else if (e.key === 'Enter' && selectedSuggestionIndex >= 0) {
                        e.preventDefault();
                        selectSuggestion(selectedSuggestionIndex);
                    } else if (e.key === 'Escape') {
                        hideSuggestions();
                    }
                });
            }
            
            document.addEventListener('click', function(e) {
                var suggestions = document.getElementById('descriptionSuggestions');
                var descInput = document.getElementById('editDescription');
                var arrow = document.querySelector('.dropdown-arrow');
                
                if (suggestions && suggestions.classList.contains('show') &&
                    e.target !== descInput && e.target !== arrow &&
                    !arrow.contains(e.target) && !suggestions.contains(e.target)) {
                    hideSuggestions();
                }
            });
        });
